\section{\texttt{LEFT JOIN}}%
%
\begin{frame}%
\frametitle{Bestellungen der Kunden}%
\begin{itemize}%
\item Wir wollen nun eine Liste unserer Kunden.%
\item<2-> Für jeden Kunden wollen wir die \sqlil{id}-Werte aller ihrer Bestellungen.%
\item<3-> Mit anderen Worten, wir wollen eine SQL-Anfrage, die zwei Spalten produziert.%
\item<4-> In der ersten Spalte -- nennen wir sie \sqlil{name} -- wollen wir die Namen der Kunden.%
\item<5-> In der zweiten Spalte -- nennen wir sie \sqlil{demand_id} -- wollen wir die \sqlil{id}-Werte der Bestellungen des Kunden.%
\item<6-> Wenn Kunden mehrere Bestellungen aufgegeben haben, dann werden sie mehrmals im Ergebnis auftauchen, jeweils mit einer Bestellung.%
\item<7-> Wenn ein Kunde keine Bestellungen aufgegeben hat, dann soll er trotzdem auftauchen, aber nur einmal, mit einem \sqlil{NULL}-Wert als \sqlil{demand_id}.%
\end{itemize}%
\end{frame}%
%
\begin{frame}%
\frametitle{Erstellen der Anfrage}%
\begin{itemize}%
\item Im ersten Schritt brauchen wir also alle Kundennamen.%
\item<2-> \sqlil{SELECT name FROM customer;} macht das.%
\item<3-> Nun müssen wir die \sqlil{id}-Werte der Tabelle~\sqlil{customer} mit der Spalte~\sqlil{customer} der Tabelle~\sqlil{demand} referenzieren.%
\item<4-> Das geht mit einem so-genannten~\sqlil{LEFT JOIN}\cite{PGDG:PD:JT}.%
\end{itemize}%
\end{frame}%
%
\begin{frame}[fragile,t]
\frametitle{\texttt{LEFT JOIN}: Syntax}%
\uncover<2-8>{%
\begin{itemize}%
\item Angenommen, Tabelle~\sqlil{Table_1} hat den Primärschlüssel~\sqlil{a}\only<-2>{.}\uncover<3->{ %
und Tabelle~\sqlil{Table_2} hat eine Spalte~\sqlil{b}, die als Fremdschlüssel auf \sqlil{Table_1.a} verweist.%
}%
\item<4-> \sqlil{SELECT ... FROM Table_1 LEFT JOIN Table_2 ON (Table_1.a = Table_2.b)} geht dann Zeile für Zeile durch \sqlil{Table_1} und sucht für jeden Wert von \sqlil{Table_1.a} alle passenden Zeilen aus \sqlil{Table_2} heraus, also alle Zeilen, für die \sqlil{Table_1.a = Table_2.b} gilt.%
\item<5-> Jede dieser Zeilen wird dann eine Zeile im Ergebnis.%
\item<6-> Gibt es keine solche Zeile, so trotzdem eine Ergebniszeile generiert, jedoch alle Werte, die vielleicht aus \sqlil{Table_2} übernommen würden, werden als~\sqlil{NULL} angegeben.%
\item<7-> Beachten Sie: Um Klarheit zu schaffen, können wir den Tabellenname als Prefix, gefolgt von einem Punkt, dem Spaltennamen voranstellen.%
\item<8-> \sqlil{Table_1.a} bedeutet also \inQuotes{Spalte~\sqlil{a} aus Tabelle~\sqlil{Table_1}}.%
\end{itemize}%
}%

\section{\texttt{INNER JOIN}}%
%
\begin{frame}%
\frametitle{Verkaufsdaten Zusammenführen}%
\begin{itemize}%
\item Wir können nun Daten von verschiedenen Tabellen zusammenführen.%
\item<2-> Erstellen wir eine Anfrage, die für jede Bestellung den Kundenname, Produktname, Preise, bestellte Anzahl, und Bestelldatum heraussucht.%
\item<3-> Das Ergebnis dieser Anfrage wäre dann eine klare und lesbare Übersicht unserer Geschäftstransaktionen.%
\item<4-> Basierend auf unserem vorigen Beispiel können wir erwarten, dass wir nun zwei~\inQuotes{Joins} brachen.%
\item<5-> Wir müssen ja Daten aus allen drei Tabellen -- \sqlil{demand}, \sqlil{customer}, und \sqlil{product} -- kombinieren.%
\item<6-> Die Anfrage wid daher etwas komplizierter werden.%
\end{itemize}%
\end{frame}%
%
\begin{frame}%
\frametitle{Grundidee}%
\begin{itemize}%
\item Wir beginnen, die Anfrage zu konstruieren.%
\item<2-> Was wir wollen ist das:~\emph{\inQuotes{Für jede Zeile in Tabelle~\sqlil{demand} brauchen wir die entsprechende Zeile in Tabelle~\sqlil{customer} und die entsprechende Zeile in Tabelle~\sqlil{product}.}}%
\item<3-> Wir wissen, dass es immer eine entsprechende Zeile in der Tabelle~\sqlil{customer} geben muss, denn die \sqlilIdx{REFERENCES}-Einschränkung erzwingt das.%
\item<3-> Wir wissen, dass es immer eine entsprechende Zeile in der Tabelle~\sqlil{product} geben muss, denn die \sqlilIdx{REFERENCES}-Einschränkung erzwingt das.%
\item<4-> Es gibt keine Zeile in \sqlil{demand} ohne entsprechende Zeile in \sqlil{customer} oder \sqlil{product}.%
\item<5-> Wir könnten wieder \sqlil{LEFT JOIN}\sqlIdx{JOIN!LEFT} verwenden, denn es könnte niemals~\sqlil{NULL} erzeugen.%
\item<6-> Lassen Sie uns aber stattdessen \sqlil{INNER JOIN}\sqlIdx{JOIN!INNER}\cite{PGDG:PD:JT} nutzen.%
\item<7-> Anders als \sqlil{LEFT JOIN} wird \sqlil{INNER JOIN} nur Zeilen ausgeben, wenn alle Tabellen zueinanderpassende Zeilen haben.%
\end{itemize}%
\end{frame}%
%
\begin{frame}[fragile,t]
\frametitle{\texttt{INNER JOIN}: Syntax}%
\uncover<2-6>{%
\begin{itemize}%
\item Angenommen, Tabelle~\sqlil{Table_1} hat den Primärschlüssel~\sqlil{a}\only<-2>{.}\uncover<3->{ %
und Tabelle~\sqlil{Table_2} hat eine Spalte~\sqlil{b}, die als Fremdschlüssel auf \sqlil{Table_1.a} verweist.%
}%
\item<4-> \sqlil{SELECT ... FROM Table_1 INNER JOIN Table_2 ON (Table_1.a = Table_2.b)} geht dann Zeile für Zeile durch \sqlil{Table_1} und sucht für jeden Wert von \sqlil{Table_1.a} alle passenden Zeilen aus \sqlil{Table_2} heraus, also alle Zeilen, für die \sqlil{Table_1.a = Table_2.b} gilt.%
\item<5-> Jede dieser Zeilen wird dann eine Zeile im Ergebnis.%
\item<6-> Gibt es keine solche Zeile, \alert{wird anders als bei \sqlil{LEFT JOIN} \emph{keine} Ergebniszeile generiert.}%
\end{itemize}%
}%

\begin{frame}%
\frametitle{Anfrage Zusammenbauen}%
\begin{itemize}%
\item Wir fangen an mit \sqlil{SELECT <something> FROM demand}.%
\item<2-> Dann schreiben wir \sqlil{INNER JOIN customer ON (customer.id = demand.customer)}.
\item<3-> Für jede Zeile in \sqlil{demand} gibt uns das die passende Zeile in \sqlil{customer}.%
\item<4-> Nun fügen wir einfach das nächste \sqlil{INNER JOIN} an, in dem wir direkt danach \sqlil{INNER JOIN product ON (product.id = demand.product)} schreiben.%
\item<5-> Damit bekommen wir die passende Zeile aus der Tabelle~\sqlil{product}.%
\item<6-> Da wir diesmal sowohl Kunden- als auch Produktnamen haben, selektieren wir \sqlil{customer.name AS customer_name} und  \sqlil{product.name AS product_name}.%
\item<7-> Wir selektieren ebenfalls \sqlil{product.price AS price}, \sqlil{demand.amount AS amount}, und \sqlil{demand.ordered as ordered}.%
\item<8-> Wir sortieren die Ausgabe mit \sqlilIdx{ORDER BY} lexikographisch nach \sqlil{customer_name}, \sqlil{ordered}, und \sqlil{product_name}, \sqlil{price}, und \sqlil{amount}.%
\end{itemize}%
\end{frame}%

\section{Sichten / Views}%
%
%
\begin{frame}%
\frametitle{Sichten / Views}%
\begin{itemize}%
\item Nun haben wir sehr schön die Verkaufsdaten unserer Fabrik aufbereitet.%
\item<2-> Wahrscheinlich brauchen wir diese Anfrage sehr häufig, wahrscheinlich jedes Mal leicht verändert.%
\item<3-> Vielleicht wollen wir nur Bestellungen aus dem aktuellen Jahr.\uncover<4->{ Dann würden wir eine \sqlil{WHERE}-Klausel anfügen.}%
\item<5-> Vielleicht wollen wir die Geldmenge pro Kunde aufaddieren.\uncover<6->{ Oder pro Produkt.}%
\item<7-> Jedes Mal würden wir eine ähnliche Abfrage schreiben.\uncover<8->{ Das ist nicht gut.}%
\item<9-> Zum Glück gibt es eine Lösung:~Wir können Anfragen als sogenannte \emph{Views}~(Sichten) speichern!\cite{PGDG:PD:CV}.%
\item<10-> Ein View funktioniert im Grunde wie eine Tabelle.%
\end{itemize}%
\end{frame}%
%
%
\begin{frame}%
\frametitle{Problem mit unserer Anfrage}%
\begin{itemize}%
\item Wir brauchen unsere Anfrage also nur als Sicht zu speichern.%
\item<2-> Als wir das tuen wollen, fällt uns ein Problem auf\only<-2>{.}\uncover<3->{Kundennamen sind nicht notwendigerweise eindeutig~(nicht~\sqlil{UNIQUE}).}%
\item<4-> Wenn wir unsere Anfrage später nutzen wollen, um Verkäufe pro Kunde aufzuaddieren, dann würden alle Verkäufe nach Kundennamen gruppiert und dann summiert.\uncover<5->{ Wenn zwei Kunden den selben Namen haben, werden ihre Bestellungen vermengt.}%
\item<6-> Mobiltelefonnummern sind aber eindeutig in unserer Datenbank.%
\item<7-> Als \sqlil{customer_name}, werden wir also \sqlil{customer.name || ', ' || customer.phone} benutzen.%
\item<8-> Der doppelten Strich~\sqlil{||} steht für das aneinanderreihen von Zeichenketten.\uncover<9->{ \sqlil{'Hello ' || 'Word!'} ergibt \sqlil{'Hello World!'}.}%
\item<10-> Problem gelöst: Da Mobiltelefonnummern eindeutig sind, sind Name + Mobiltelefonnummer auch eindeutig.%
\end{itemize}%
\end{frame}%
%
%
\begin{frame}%
\frametitle{Anfrage als Sicht speichern und benutzen}%
\begin{itemize}%
\item Wir können unsere Anfrage nun als Sicht speichern.%
\item<2-> Das geht sehr einfach: Wir schreiben~\sqlil{CREATE VIEW sale AS ...}, wobei \sqlil{...} durch unsere Anfrage zu ersetzen ist.
\item<3-> Damit wird unsere Anfrage unter dem Namen~\sqlil{sale} gespeichert.%
\item<4-> Sie kann dann wie eine Tabelle verwendet werden.%
\item<5-> Wir können schreiben \sqlil{SELECT * from sale;}.%
\end{itemize}%
\end{frame}%

\section{Die Sicht Benutzen}%
%
\begin{frame}%
\frametitle{Benutzen unserer neuen Sicht}%
\begin{itemize}%
\item Benutzen wir nun unsere neue Sicht, um folgende Fragen zu beantworten\uncover<2->{:%
\begin{itemize}%
\item Welcher Kunde hat den meisten Umsatz gemacht?%
\item<3-> Welches Produkt hat den meisten Umsatz gemacht?%
\end{itemize}%
}%
\item<4-> Wir können den Gesamtpreis jeder Bestellung einfach ausrechnen.\uncover<5->{ Wir kennen ja die Anzahl der bestellten Einheiten~(\sqlil{amount}) und den Preis pro Einheit~(\sqlil{price}).}%
\item<5-> Pro Zeile müssen wir also einfach~\sqlil{amount * price} berechnen.%
\item<6-> Da wir die Summe dieser Werte pro Kunde bzw.\ pro Produkt wollen, müssen wir die Zeilen jeweils nach \sqlil{customer_name} oder \sqlil{product_name} gruppieren.%
\item<7-> Pro Gruppe berechnen wir dann~\sqlil{SUM(amount * price)}\cite{PGDG:PD:AF}.%
\item<8-> Schließlich sortieren wir die Werte absteigend, weil wir ja die Bestseller am Anfang haben wollen.%
\end{itemize}%
\end{frame}%

\section{Zusammenfassung}%
%
\begin{frame}%
\frametitle{Zusammenfassung}%
\begin{itemize}%
\item Wir sind nun ein ganzes Stück stärker geworden.%
\item<2-> Mit \sqlil{JOIN}s sind wir nun in der Lage, Daten aus verschiedenen Tabellen zusammenzufügen.%
\item<3-> Das ist eines der wichtigsten Features, das Datenbanken uns bieten.%
\item<4-> Wir haben auch gelernt, wie wir Sichten~(Views) erstellen können, also gespeicherte Anfragen.%
\item<5-> Damit ist es uns nicht nur möglich, bestimmte, oft benötigte Anfragen zu speichern um unnötiges, mehrmaliges eingeben zu verhindern.%
\item<6-> Sichten sind auch ein zentrales Werkzeug, um verschiedenen Werkzeugen und Benutzern {\dots} nun {\dots} verschiedene Sichten auf die Daten zu bieten.%
\item<7-> Wir können sie nutzen, um bestimmten Nutzern oder Applikationen nur bestimmte Teile unserer gespeicherten Daten zugänglich zu machen.%
\end{itemize}%
\end{frame}%
